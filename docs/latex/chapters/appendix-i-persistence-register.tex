\subsection{Relational Table Definitions}
\begin{enumerate}
    \item \textbf{\nolinkurl{hie_fhir_resources}}: Primary key \nolinkurl{(resource_type, resource_id, version_id)}. Stores raw JSON resource payload, optimistic lock version, last-updated timestamp, and JSONB search indexes.
    \item \textbf{\nolinkurl{hie_operations_resources}}: Primary key \nolinkurl{(operation_id, sequence_id)}. Stores \nolinkurl{Pragma} state envelopes, task checkpoint progressions, and execution latencies.
    \item \textbf{\nolinkurl{agora_resource_mappings}}: Primary key UUID \nolinkurl{id}. Maps Harmonia resource references to Matrix room/space IDs with \nolinkurl{status} and timestamp columns.
    \item \textbf{\nolinkurl{agora_as_transactions}}: Primary key \nolinkurl{transaction_id}. Enforces AS push transaction idempotency with receive/process timestamps.
\end{enumerate}
